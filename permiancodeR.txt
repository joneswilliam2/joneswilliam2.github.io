library(tidyverse)
library(plm)
library(lmtest)
library(sandwich)
library(ivreg)   # IV estimation
library(stargazer)
library(fixest)   # county-specific trends

# 1 Load in Data
df <- read_csv("data1.csv") %>%
  select(year, county, class,
         teenbirths, medHHincome1k, assocorhigher,
         povertypercent, drugPoisRate, singleParents,
         percentDisYouth, unemp, crime100k, RESpopulation) %>%
  # Drop things2doPC — time-invariant, absorbed by FE, weak construction
  # Retained qualitatively in Section 5.4
  mutate(
    log_income   = log(medHHincome1k),
    log_pop      = log(RESpopulation),
    permian_flag = as.integer(class == "Permian")
  )

# 2 WTI Oil Price (2009-2021)
# Annual average WTI crude oil price, USD per barrel
# Source: EIA
wti <- tibble(
  year = 2009:2021,
  wti_price = c(61.95, 79.48, 94.88, 94.05, 97.98,
                93.17, 48.66, 43.29, 50.80, 65.23,
                56.99, 39.57, 68.17)
)

df <- df %>% left_join(wti, by = "year")

# instrument
df <- df %>%
  mutate(wti_permian = wti_price * permian_flag)

# 3
# teenbirths available through 2020
# drugPoisRate available through 2021
df_teen <- df %>% filter(!is.na(teenbirths))   # 2009-2020
df_drug <- df %>% filter(!is.na(drugPoisRate)) # 2009-2021

pdata_teen <- pdata.frame(df_teen, index = c("county", "year"))
pdata_drug <- pdata.frame(df_drug, index = c("county", "year"))

# model 1: baseline TWFE (main spec)
m1 <- plm(teenbirths ~ medHHincome1k + assocorhigher +
            povertypercent + singleParents + percentDisYouth,
          data   = pdata_teen,
          model  = "within",
          effect = "twoways")

# model 2: drugpoisoning rate added
m2 <- plm(teenbirths ~ medHHincome1k + assocorhigher +
            povertypercent + singleParents + percentDisYouth +
            drugPoisRate,
          data   = pdata_teen %>% filter(!is.na(drugPoisRate)),
          model  = "within",
          effect = "twoways")

# model 3: log income instead of levels
m3 <- plm(teenbirths ~ log_income + assocorhigher +
            povertypercent + singleParents + percentDisYouth,
          data   = pdata_teen,
          model  = "within",
          effect = "twoways")

# cluster robust county SEs 
se_m1 <- coeftest(m1, vcov = vcovHC(m1, type = "HC1", cluster = "group"))
se_m2 <- coeftest(m2, vcov = vcovHC(m2, type = "HC1", cluster = "group"))
se_m3 <- coeftest(m3, vcov = vcovHC(m3, type = "HC1", cluster = "group"))

print(se_m1)
print(se_m2)
print(se_m3)

# SPHS mediator check
# model 1 excluding single-parent share
m1_nosp <- plm(teenbirths ~ medHHincome1k + assocorhigher +
                 povertypercent + percentDisYouth,
               data   = pdata_teen,
               model  = "within",
               effect = "twoways")
se_m1_nosp <- coeftest(m1_nosp, vcov = vcovHC(m1_nosp, type = "HC1", cluster = "group"))
print(se_m1_nosp)

# SPHS as the outcome: asking incomes effect on singleparents
m_sp <- plm(singleParents ~ medHHincome1k + assocorhigher +
              povertypercent + percentDisYouth,
            data   = pdata_teen,
            model  = "within",
            effect = "twoways")
se_m_sp <- coeftest(m_sp, vcov = vcovHC(m_sp, type = "HC1", cluster = "group"))
print(se_m_sp)

# parallel trends
# TWFE assumes no county-specific trends correlated with income.
# checking county specific linear trends + leads test

# county-specific linear time trends
df_trend <- df_teen %>%
  mutate(t = year - min(year))

m1_trend <- feols(teenbirths ~ medHHincome1k + assocorhigher +
                    povertypercent + singleParents + percentDisYouth |
                    county[t] + year,
                  data    = df_trend,
                  cluster = ~county)
summary(m1_trend)

# leads test: does future income predict teen births (reverse causality)
df_lead <- df %>%
  arrange(county, year) %>%
  group_by(county) %>%
  mutate(inc_lead = dplyr::lead(medHHincome1k)) %>%
  ungroup() %>%
  filter(!is.na(teenbirths), !is.na(inc_lead))

pdata_lead <- pdata.frame(df_lead, index = c("county", "year"))

m1_lead <- plm(teenbirths ~ medHHincome1k + inc_lead + assocorhigher +
                 povertypercent + singleParents + percentDisYouth,
               data   = pdata_lead,
               model  = "within",
               effect = "twoways")
se_m1_lead <- coeftest(m1_lead, vcov = vcovHC(m1_lead, type = "HC1", cluster = "group"))
print(se_m1_lead)

# diagnostic: how much does a linear trend explain?
trend_share <- function(v) {
  d <- df_trend %>% filter(!is.na(.data[[v]]))
  ss_fe <- 0
  ss_tr <- 0
  for (cty in unique(d$county)) {
    g <- d %>% filter(county == cty)
    ss_fe <- ss_fe + sum((g[[v]] - mean(g[[v]]))^2)
    ss_tr <- ss_tr + sum(resid(lm(g[[v]] ~ g$t))^2)
  }
  1 - ss_tr / ss_fe
}
sapply(c("teenbirths", "medHHincome1k", "singleParents"), trend_share)

# IV
# instrument = WTI * Permian flag for income endogeneity

df_iv <- df_teen %>%
  mutate(
    county_f = factor(county),
    year_f   = factor(year)
  ) %>%
  filter(!is.na(teenbirths), !is.na(wti_price))

iv_m1 <- ivreg(
  teenbirths ~ medHHincome1k + assocorhigher +
    povertypercent + singleParents + percentDisYouth +
    county_f + year_f |
    wti_permian + assocorhigher +
    povertypercent + singleParents + percentDisYouth +
    county_f + year_f,
  data = df_iv
)

# diagnostics
summary(iv_m1, diagnostics = TRUE)

# SEs clustered by county
vcov_iv <- vcovCL(iv_m1, cluster = ~county)
coeftest(iv_m1, vcov = vcov_iv)[c("medHHincome1k", "singleParents"), ]

# 7 robustness: drug poisoning as outcome
m_drug <- plm(drugPoisRate ~ medHHincome1k + assocorhigher +
                povertypercent + singleParents + percentDisYouth +
                unemp,
              data   = pdata_drug,
              model  = "within",
              effect = "twoways")

se_drug <- coeftest(m_drug, vcov = vcovHC(m_drug, type = "HC1", cluster = "group"))
print(se_drug)

# 8 heterogeneiety by county type
df_het <- df_teen %>%
  mutate(
    permian_income = medHHincome1k * permian_flag,
    county_f = factor(county),
    year_f   = factor(year)
  )

m_het <- lm(teenbirths ~ medHHincome1k + permian_income +
              assocorhigher + povertypercent + singleParents +
              percentDisYouth + county_f + year_f,
            data = df_het)

# SEs clustered by county
coeftest(m_het, vcov = vcovCL(m_het, cluster = ~county))[c("medHHincome1k", "permian_income"), ]

# 9 exports result table TEX
stargazer(m1, m2, m3,
          se    = list(se_m1[, 2], se_m2[, 2], se_m3[, 2]),
          p     = list(se_m1[, 4], se_m2[, 4], se_m3[, 4]),
          type  = "latex",
          title = "Two-Way Fixed Effects: Predictors of Teen Birth Rate",
          dep.var.labels = "Teen Birth Rate (per 1,000 females 15-19)",
          covariate.labels = c("Median HH Income ($k)",
                               "Log Median HH Income",
                               "Assoc. Degree or Higher (%)",
                               "Poverty Rate (%)",
                               "Single Parent HH (%)",
                               "Disconnected Youth (%)",
                               "Drug Poisoning Rate"),
          add.lines    = list(c("County FE", "Yes", "Yes", "Yes"),
                              c("Year FE",   "Yes", "Yes", "Yes")),
          notes        = "Cluster-robust SEs at county level.",
          out          = "results_table.tex")

cat("\nDone. Check results_table.tex for LaTeX output.\n")

# theme
theme_permian <- function() {
  theme_classic(base_size = 13, base_family = "serif") +
    theme(
      plot.title    = element_text(face = "bold", size = 13),
      axis.title    = element_text(size = 12),
      axis.text     = element_text(size = 11),
      legend.title  = element_blank(),
      legend.text   = element_text(size = 11),
      legend.position = "bottom",
      plot.caption  = element_text(size = 9, hjust = 0, color = "gray40")
    )
}
